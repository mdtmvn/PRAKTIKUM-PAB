            tombolKategori(context, "Fiksi & Novel"),

  Widget tombolKategori(BuildContext context, String text) {
    return Padding(
      padding: const EdgeInsets.all(6),
      child: SizedBox(
        width: 220,
        child: ElevatedButton(
          onPressed: () {
            Navigator.push(

              MaterialPageRoute(builder: (_) => const DetailBukuPage()),
            );
          },
          child: Text(text),
        ),
      ),
    );
  }
